## 9. Interactive Lead Capture Form (React JSX Code)

```jsx
import React, { useState } from 'react';

export default function LeadCaptureForm() {
  const [formData, setFormData] = useState({
    fullName: '',
    email: '',
    phone: '',
    companyUrl: '',
    serviceCategory: 'Google Ads Optimization',
    region: 'Bangalore (SAB)',
    budgetRange: '₹50,000 - ₹1,00,000/mo',
    projectNotes: '',
    website_url: '' // Spambot honeypot
  });

  const [status, setStatus] = useState('idle');

  const handleSubmit = async (e) => {
    e.preventDefault();
    if (formData.website_url) {
      // Silent rejection for spambots
      setStatus('success');
      return;
    }
    setStatus('submitting');
    try {
      const res = await fetch('/api/leads', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(formData)
      });
      if (res.ok) {
        setStatus('success');
      } else {
        setStatus('error');
      }
    } catch (err) {
      setStatus('error');
    }
  };

  return (
    <section className="py-12 bg-slate-900 text-white rounded-xl p-8 max-w-4xl mx-auto my-8 border border-slate-800">
      <h2 className="text-3xl font-bold mb-4 text-center">Schedule Your Bangalore Growth Engineering Consultation</h2>
      <p className="text-slate-400 mb-8 text-center max-w-2xl mx-auto">
        Direct technical access. Zero agency proxies. Get a comprehensive performance audit and strategy proposal within 24 hours.
      </p>

      {status === 'success' ? (
        <div className="bg-emerald-950 border border-emerald-500 text-emerald-200 p-6 rounded-lg text-center">
          <h3 className="text-xl font-bold mb-2">Inquiry Received Successfully!</h3>
          <p>Vraj Vithalani will review your application and respond within 24 business hours.</p>
        </div>
      ) : (
        <form onSubmit={handleSubmit} className="space-y-6">
          {/* Honeypot Field */}
          <input 
            type="text" 
            name="website_url" 
            value={formData.website_url} 
            onChange={(e) => setFormData({...formData, website_url: e.target.value})} 
            style={{ display: 'none' }} 
            tabIndex="-1" 
            autocomplete="off" 
          />

          <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
            <div>
              <label className="block text-sm font-medium mb-2 text-slate-300">Full Name *</label>
              <input 
                type="text" 
                required 
                className="w-full bg-slate-800 border border-slate-700 rounded-lg p-3 text-white focus:outline-none focus:border-blue-500"
                value={formData.fullName} 
                onChange={(e) => setFormData({...formData, fullName: e.target.value})} 
              />
            </div>
            <div>
              <label className="block text-sm font-medium mb-2 text-slate-300">Work Email *</label>
              <input 
                type="email" 
                required 
                className="w-full bg-slate-800 border border-slate-700 rounded-lg p-3 text-white focus:outline-none focus:border-blue-500"
                value={formData.email} 
                onChange={(e) => setFormData({...formData, email: e.target.value})} 
              />
            </div>
          </div>

          <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
            <div>
              <label className="block text-sm font-medium mb-2 text-slate-300">Phone / WhatsApp *</label>
              <input 
                type="tel" 
                required 
                className="w-full bg-slate-800 border border-slate-700 rounded-lg p-3 text-white focus:outline-none focus:border-blue-500"
                value={formData.phone} 
                onChange={(e) => setFormData({...formData, phone: e.target.value})} 
              />
            </div>
            <div>
              <label className="block text-sm font-medium mb-2 text-slate-300">Company Website / App URL</label>
```
